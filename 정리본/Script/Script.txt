    internal sealed class SubwayGeometry
    {
        readonly List<Vector3> vertices=new List<Vector3>();
        readonly List<Vector3> normals=new List<Vector3>();
        readonly Dictionary<int,List<int>> faces=new Dictionary<int,List<int>>();
        readonly Material[] paints;
        readonly string meshFolder;
        static readonly Vector3[] axes={Vector3.right,Vector3.up,Vector3.forward};
        public SubwayGeometry(Material[] materials,string directory) {paints=materials;meshFolder=directory;}
        public void Box(float x,float y,float z,float sx,float sy,float sz,int mat)
        {
            var center=new Vector3(x,y,z);var half=new Vector3(sx,sy,sz)*.5f;
            if(!faces.TryGetValue(mat,out var triangles)) faces.Add(mat,triangles=new List<int>());
            for(int i=0;i<3;i++) for(int sign=-1;sign<=1;sign+=2)
            {
                var n=axes[i]*sign; var origin=center+Vector3.Scale(n,half);
                var u=Vector3.Scale(axes[(i+1)%3],half);var v=Vector3.Scale(axes[(i+2)%3],half)*sign;
                int first=vertices.Count;
                vertices.Add(origin-u-v);vertices.Add(origin+u-v);vertices.Add(origin+u+v);vertices.Add(origin-u+v);
                for(int j=0;j<4;j++) normals.Add(n);
                triangles.Add(first);triangles.Add(first+1);triangles.Add(first+2);
                triangles.Add(first);triangles.Add(first+2);triangles.Add(first+3);
            }
        }
        public void Flush(Transform parent,string label)
        {
            if(vertices.Count==0)return;
            var mesh=new Mesh {name=label,indexFormat=vertices.Count>65535?IndexFormat.UInt32:IndexFormat.UInt16};
            mesh.SetVertices(vertices);mesh.SetNormals(normals);
            int[] slots=faces.Keys.OrderBy(k=>k).ToArray();mesh.subMeshCount=slots.Length;
            for(int i=0;i<slots.Length;i++) mesh.SetTriangles(faces[slots[i]],i,false);
            mesh.RecalculateBounds();
            AssetDatabase.CreateAsset(mesh,meshFolder+"/"+label+".asset");
            var g=new GameObject(label,typeof(MeshFilter),typeof(MeshRenderer));g.transform.SetParent(parent,false);
            g.GetComponent<MeshFilter>().sharedMesh=mesh;
            var renderer=g.GetComponent<MeshRenderer>();renderer.sharedMaterials=slots.Select(i=>paints[i]).ToArray();
            renderer.shadowCastingMode=ShadowCastingMode.Off;
            vertices.Clear();normals.Clear();faces.Clear();
        }
    }

using UnityEngine;

namespace LineFour
{
    public sealed class RunState
    {
        public static readonly float[] Stops = { 0, 540, 1130 };
        public static readonly string[] Names = { "혜화", "동대문", "동대문역사문화공원" };
        public float Distance, Speed, Dwell;
        public int Target = 1, Served;
        public bool Doors, Complete;
        public bool InZone => Mathf.Abs(Stops[Target] - Distance) <= 4;
        public bool Stopped => Speed < 0.01f;

        public void Step(float dt, bool power, bool brake, bool emergency)
        {
            if (Complete) return;
            if (Doors) { Dwell += dt; return; }
            float previous = Speed;
            float acceleration = emergency ? -2.15f : brake ? -1.23f :
                power ? Mathf.Lerp(1.02f, 0.48f, Speed / 18) : -0.09f;
            Speed = Mathf.Clamp(Speed + acceleration * dt, 0, 18);
            Distance += (previous + Speed) * 0.5f * dt;
            if (Distance >= Stops[Target] + 26) { Distance = Stops[Target] + 26; Speed = 0; }
        }

        public void Service()
        {
            if (Complete || !Stopped || !InZone) return;
            if (!Doors) { Doors = true; Dwell = 0; return; }
            if (Dwell < 3) return;
            Doors = false;
            Served = Target;
            Complete = Target == Stops.Length - 1;
            if (!Complete) Target++;
        }

        public void RestartSection()
        {
            if (Complete) { Served = 0; Target = 1; }
            Distance = Stops[Served]; Speed = 0; Dwell = 0; Doors = false; Complete = false;
        }
    }

        AudioClip MakeSound(bool bell)
        {
            const int rate = 22050;
            float[] samples = new float[rate];
            for (int i=0;i<rate;i++)
            {
                float t=(float)i/rate;
                if (bell) {
                    float local=t<.35f?t:t-.35f, hz=t<.35f?660:880;
                    samples[i]=.4f*Mathf.Sin(2*Mathf.PI*hz*local)*Mathf.Exp(-7*local)*Mathf.Clamp01(local*140);
                } else {
                    samples[i]=.34f*Mathf.Sin(2*Mathf.PI*68*t)+.12f*Mathf.Sin(2*Mathf.PI*136*t)
                        +.11f*Mathf.Sin(2*Mathf.PI*343*t)*Mathf.Pow(.5f+.5f*Mathf.Cos(2*Mathf.PI*4*t),12);
                }
            }
            var clip=AudioClip.Create(bell?"Platform chime":"Traction loop", rate,1,rate,false);
            clip.SetData(samples,0); return clip;

    internal sealed class SubwayGeometry
    {
        readonly List<Vector3> vertices=new List<Vector3>();
        readonly List<Vector3> normals=new List<Vector3>();
        readonly Dictionary<int,List<int>> faces=new Dictionary<int,List<int>>();
        readonly Material[] paints;
        readonly string meshFolder;
        static readonly Vector3[] axes={Vector3.right,Vector3.up,Vector3.forward};
        public SubwayGeometry(Material[] materials,string directory) {paints=materials;meshFolder=directory;}
        public void Box(float x,float y,float z,float sx,float sy,float sz,int mat, Quaternion? rotation=null)
        {
            var center=new Vector3(x,y,z);var half=new Vector3(sx,sy,sz)*.5f;
            if(!faces.TryGetValue(mat,out var triangles)) faces.Add(mat,triangles=new List<int>());
            for(int i=0;i<3;i++) for(int sign=-1;sign<=1;sign+=2)
            {
                var q=rotation??Quaternion.identity;
                var n=q*(axes[i]*sign); var origin=center+n*half[i];
                var u=q*Vector3.Scale(axes[(i+1)%3],half);var v=q*Vector3.Scale(axes[(i+2)%3],half)*sign;
                int first=vertices.Count;
                vertices.Add(origin-u-v);vertices.Add(origin+u-v);vertices.Add(origin+u+v);vertices.Add(origin-u+v);
                for(int j=0;j<4;j++) normals.Add(n);
                triangles.Add(first);triangles.Add(first+1);triangles.Add(first+2);
                triangles.Add(first);triangles.Add(first+2);triangles.Add(first+3);
            }
        }
        public void Beam(Vector3 a,Vector3 b,float width,float depth,int mat)
        {
            Vector3 mid=(a+b)*.5f;
            Box(mid.x,mid.y,mid.z,width,depth,(b-a).magnitude,mat,Quaternion.FromToRotation(Vector3.forward,(b-a).normalized));
        }
        public void Soffit(float left,float right,float y,float start,float end,int mat)
        {
            if(!faces.TryGetValue(mat,out var tris))faces.Add(mat,tris=new List<int>());
            int n=vertices.Count;
            vertices.AddRange(new[]{new Vector3(left,y,start),new Vector3(right,y,start),new Vector3(right,y,end),new Vector3(left,y,end)});
            for(int i=0;i<4;i++)normals.Add(Vector3.down);
            tris.AddRange(new[]{n,n+1,n+2,n,n+2,n+3});
        }
        public void Flush(Transform parent,string label)
        {
            if(vertices.Count==0)return;
            var mesh=new Mesh {name=label,indexFormat=vertices.Count>65535?IndexFormat.UInt32:IndexFormat.UInt16};
            mesh.SetVertices(vertices);mesh.SetNormals(normals);
            int[] slots=faces.Keys.OrderBy(k=>k).ToArray();mesh.subMeshCount=slots.Length;
            for(int i=0;i<slots.Length;i++) mesh.SetTriangles(faces[slots[i]],i,false);
            mesh.RecalculateBounds();
            var reusable=(label.StartsWith("Track_")||label.StartsWith("Wayside_"))?
                AssetDatabase.LoadAssetAtPath<Mesh>("Assets/OidoLine4/RouteAssets/"+label+".asset"):null;
            if(reusable) { UnityEngine.Object.DestroyImmediate(mesh); mesh=reusable; }
            else AssetDatabase.CreateAsset(mesh,meshFolder+"/"+label+".asset");
            var g=new GameObject(label,typeof(MeshFilter),typeof(MeshRenderer));g.transform.SetParent(parent,false);
            g.GetComponent<MeshFilter>().sharedMesh=mesh;
            var renderer=g.GetComponent<MeshRenderer>();renderer.sharedMaterials=slots.Select(i=>paints[i]).ToArray();
            renderer.shadowCastingMode=ShadowCastingMode.Off;
            vertices.Clear();normals.Clear();faces.Clear();
        }
    }

                r.SetBlendShapeWeight(shape,0);

    public enum DoorPhase { Closed, Opening, Boarding, Closing }
    public sealed class RunState
    {
        public static readonly float[] Stops = RouteSchedule.Stops;
        public static readonly string[] Names = RouteSchedule.Names;
        public const float Tolerance=4;
        public int Score, Misses, LastAccuracy;
        public float LastError;
        public string Receipt="첫 승하차 대기";
        public bool Missed;
        public float Distance, Speed, Dwell, DoorTravel;
        public int Target, Served = -1;
        public DoorPhase Phase;
        public bool Doors => Phase!=DoorPhase.Closed;
        public bool Complete;
        public bool InZone => Mathf.Abs(Stops[Target] - Distance) <= Tolerance;
        public bool Stopped => Speed < 0.01f;

        public void Step(float dt, bool power, bool brake, bool emergency)
        {
            if (Complete || Missed) return;
            if (Phase==DoorPhase.Opening) {
                DoorTravel=Mathf.Min(1,DoorTravel+dt/1.6f);
                if(DoorTravel>=1) { Phase=DoorPhase.Boarding; Score+=25;Receipt=Names[Target]+"  문 열기 +25"; }
                return;
            }
            if (Phase==DoorPhase.Boarding) { Dwell+=dt; return; }
            if (Phase==DoorPhase.Closing) {
                DoorTravel=Mathf.Max(0,DoorTravel-dt/1.8f);
                if(DoorTravel==0) {
                    Phase=DoorPhase.Closed; Served=Target;
                    LastError=Distance-Stops[Target];
                    LastAccuracy=Mathf.RoundToInt(100*Mathf.Clamp01(1-Mathf.Abs(LastError)/Tolerance));
                    Score+=225+LastAccuracy;
                    Receipt=Names[Target]+"  승하차 +200  ·  정차 +"+LastAccuracy+"  ·  닫기 +25";
                    Complete=Target==Stops.Length-1;
                    if(!Complete) Target++;
                }
                return;
            }
            if (Served<0 || (Stopped && InZone)) {Speed=0;return;}
            float previous = Speed;
            float acceleration = emergency ? -2.15f : brake ? -1.23f :
                power ? Mathf.Lerp(1.02f, 0.48f, Speed / 18) : -0.09f;
            Speed = Mathf.Clamp(Speed + acceleration * dt, 0, 18);
            Distance += (previous + Speed) * 0.5f * dt;
            if (Distance > Stops[Target]+Tolerance) {
                Score-=100;Misses++;Missed=true;Speed=0;Receipt=Names[Target]+"  무정차 통과 −100";
            }
        }

        public void Service()
        {
            if (Complete || Missed || !Stopped || !InZone) return;
            if (Phase==DoorPhase.Closed) { Phase=DoorPhase.Opening; Dwell=0; Speed=0; }
            else if (Phase==DoorPhase.Boarding && Dwell>=3) Phase=DoorPhase.Closing;
        }

        public void RestartSection()
        {
            if(Doors)return;
            if (Complete) { Served=-1; Score=Misses=LastAccuracy=0;LastError=0;Receipt="첫 승하차 대기"; }
            Target=Served+1; Distance=Stops[Mathf.Max(0,Served)]; Speed=0;
            if(Missed) Distance=Stops[Target]-24;
            Missed=false;
            DoorTravel=0; Dwell=0; Phase=DoorPhase.Closed; Complete=false;
        }
    }

        AudioClip MakeSound(bool bell)
        {
            const int rate = 22050;
            float[] samples = new float[rate];
            for (int i=0;i<rate;i++)

                float t=(float)i/rate;
                if (bell) {
                    float local=t<.35f?t:t-.35f, hz=t<.35f?660:880;
                    samples[i]=.4f*Mathf.Sin(2*Mathf.PI*hz*local)*Mathf.Exp(-7*local)*Mathf.Clamp01(local*140);
                } else {
                    samples[i]=.34f*Mathf.Sin(2*Mathf.PI*68*t)+.12f*Mathf.Sin(2*Mathf.PI*136*t)

   CGPROGRAM
   #pragma vertex vert
   #pragma fragment frag
   #pragma multi_compile_fog
   #include "UnityCG.cginc"
   struct Input { float4 vertex:POSITION; float3 normal:NORMAL; fixed4 color:COLOR; };
   struct Varying { float4 pos:SV_POSITION; fixed4 color:COLOR; UNITY_FOG_COORDS(1) };
   Varying vert(Input v) {
    Varying o; o.pos=UnityObjectToClipPos(v.vertex);
    o.color=v.color*(.62+.38*saturate(dot(UnityObjectToWorldNormal(v.normal),normalize(float3(-.3,.8,.25)))));
    UNITY_TRANSFER_FOG(o,o.pos); return o;
   }
   fixed4 frag(Varying i):SV_Target { fixed4 c=i.color; UNITY_APPLY_FOG(i.fogCoord,c); return c; }
   ENDCG
  }

using System.Collections.Generic;

namespace LineFour.Oido

    // Three cohorts retain the people who just alighted until the train leaves the platform.
    public sealed class PassengerFlow : MonoBehaviour
    {
        const int Count=14;
        sealed class Person { public Transform root,left,right; }
        Person[,] people=new Person[3,Count];
        float[] doors;
        Transform train;
        Material cloth;
        List<Mesh> meshes=new List<Mesh>();
        public int VisibleCount {get; private set;}
        public int WalkingCount {get; private set;}
        public void Prepare(float[] centers,Transform consist,Shader fabric)
        {
            doors=centers;train=consist;
            if(doors==null||doors.Length<4)return;
            cloth=new Material(fabric);
            string[] colors={"356175","AB644D","55634A","C18B47","826583","374C78","9B505C"};
            Mesh leg=Part("Trousers",new[]{new Shape(0,-.33f,0,.145f,.69f,.185f,"343C49"),new Shape(0,-.665f,.046f,.156f,.12f,.25f,"242932")});
            for(int c=0;c<3;c++)for(int j=0;j<Count;j++){
                int k=(j+c*3)%colors.Length;
                if(meshes.Count<=1+k){
                    string skin=new[]{"C88968","E2B699","AE795F"}[k%3],coat=colors[k];
                    Part("Commuter "+k,new[]{
                        new Shape(0,1.02f,0,.43f,.66f,.31f,coat),new Shape(0,1.42f,0,.25f,.25f,.23f,skin),
                        new Shape(0,1.56f,.013f,.235f,.315f,.245f,skin),new Shape(0,1.66f,-.02f,.252f,.18f,.253f,k%2==0?"322F30":"503C34"),
                        new Shape(-.227f,1.04f,0,.112f,.53f,.14f,coat),new Shape(.227f,1.04f,0,.112f,.53f,.14f,coat),
                        new Shape(-.23f,.776f,.016f,.094f,.13f,.10f,skin),new Shape(.23f,.776f,.016f,.094f,.13f,.10f,skin),
                        new Shape(0,1.03f,-.164f,.26f,.38f,.14f,k%2==0?"444348":"826548")});
                }
                var p=new Person();p.root=new GameObject("Passenger "+c+"-"+j).transform;p.root.SetParent(transform,false);
                Piece(p.root,meshes[1+k],Vector3.zero);
                p.left=Piece(p.root,leg,new Vector3(-.094f,.735f,0));p.right=Piece(p.root,leg,new Vector3(.094f,.735f,0));
                p.root.localScale=Vector3.one*(.91f+.03f*((c+j)%6));people[c,j]=p;
            }
        }
        struct Shape {
            public Vector3 center,size;public Color color;
            public Shape(float x,float y,float z,float w,float h,float d,string rgb) {
                center=new Vector3(x,y,z);size=new Vector3(w,h,d);ColorUtility.TryParseHtmlString("#"+rgb,out color);
            }
        }
        Mesh Part(string label,Shape[] shapes)
        {
            var verts=new List<Vector3>();var colors=new List<Color>();var tris=new List<int>();
            foreach(var s in shapes){
                int first=verts.Count;
                float[] radius={.56f,.94f,1,.62f};float[] ys={-.5f,-.36f,.29f,.5f};
                for(int row=0;row<4;row++)for(int a=0;a<8;a++){
                    float angle=2*Mathf.PI*a/8;verts.Add(s.center+Vector3.Scale(s.size,new Vector3(.5f*radius[row]*Mathf.Cos(angle),ys[row],.5f*radius[row]*Mathf.Sin(angle))));colors.Add(s.color);
                }
                for(int row=0;row<3;row++)for(int a=0;a<8;a++){
                    int n=first+row*8+a,m=first+row*8+(a+1)%8;
                    tris.AddRange(new[]{n,n+8,m,m,n+8,m+8});

                Vector3 outside=new Vector3(-3.85f,1.265f,stop+dz-1.15f);
                Vector3 start=incoming?wait:inside,end=incoming?inside:outside;
                bool moving=q>0&&q<1&&!last;float swing=moving?Mathf.Sin(q*Mathf.PI*6)*23:0;
                if(moving)WalkingCount++;
                Vector3 pos=last?new Vector3(-3.85f,1.265f,previous+dz-1.15f):
                    q<.5f?Vector3.Lerp(start,doorway,q*2):Vector3.Lerp(doorway,end,(q-.5f)*2);
                pos.y+=Mathf.Abs(swing)*.0008f;
                p.root.SetPositionAndRotation(pos,Quaternion.Euler(0,last?-65:incoming?88:-92,0));
                p.left.localRotation=Quaternion.Euler(swing,0,0);p.right.localRotation=Quaternion.Euler(-swing,0,0);
            }
        }
        void OnDestroy(){foreach(var mesh in meshes)Destroy(mesh);if(cloth)Destroy(cloth);}
    }
